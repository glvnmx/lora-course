для расчётов и LoraConfig
from peft import LoraConfig

def lora_params(d_in, d_out, r, modules=1):
    return modules * r * (d_in + d_out)

config = LoraConfig(
    r=16,
    lora_alpha=32,
    lora_dropout=0.05,
    target_modules=["q_proj", "v_proj"],
    task_type="CAUSAL_LM",
)
print(lora_params(4096, 4096, config.r, modules=2))
